  <script>
    // Global State
    let state = {
      keywords: [],
      templates: []
    };

    // Color palette for keyword tiles
    const tileColors = ['bg-emerald-500', 'bg-sky-500', 'bg-amber-500', 'bg-fuchsia-500', 'bg-rose-500', 'bg-violet-500'];

    // UI Utility Functions
    const utils = {
      showLoader() {
        document.getElementById('loader').classList.remove('hidden');
      },
      
      hideLoader() {
        document.getElementById('loader').classList.add('hidden');
      },
      
      showMessage(message, isError = false) {
        const toast = document.getElementById('toast');
        const toastMessage = document.getElementById('toast-message');
        
        toast.className = `fixed top-4 left-4 right-4 max-w-lg mx-auto z-40 px-6 py-4 rounded-lg shadow-lg transform transition-all duration-300 ${isError ? 'bg-red-500' : 'bg-green-500'}`;
        toastMessage.textContent = message;
        toast.classList.remove('hidden');
        
        setTimeout(() => {
          toast.classList.add('hidden');
        }, 3000);
      }
    };

    // Backend Communication Helper
    function serverCall(functionName, args, callback) {
      utils.showLoader();
      
      // Simulating server call with dummy data since we're not actually connected to Google Apps Script
      setTimeout(() => {
        utils.hideLoader();
        
        if (functionName === 'getInitialData') {
          const dummyData = {
            keywords: ["urgent", "payment", "help", "issue", "subscribe", "question", "refund", "support"],
            templates: [
              "Thank you for contacting us. We'll get back to you within 24 hours.",
              "We appreciate your patience. Your request is being processed.",
              "For urgent matters, please call our hotline at 1-800-HELP.",
              "Your payment has been received and confirmed.",
              "We've updated your subscription. Changes will take effect immediately.",
              "Our team is investigating this issue. We'll update you shortly.",
              "Please provide more details so we can assist you better.",
              "Thank you for your feedback. We value your input.",
              "Your refund has been initiated and will be processed in 3-5 business days.",
              "Is there anything else we can help you with today?"
            ]
          };
          callback(dummyData);
          utils.showMessage('Data loaded successfully');
        } else if (functionName === 'addKeyword') {
          callback({ success: true });
          utils.showMessage('Keyword added successfully');
        } else if (functionName === 'deleteKeyword') {
          callback({ success: true });
          utils.showMessage('Keyword deleted successfully');
        } else if (functionName === 'saveTemplates') {
          callback({ success: true });
          utils.showMessage('Templates saved successfully');
        }
      }, 500);
      
      // Uncomment below for actual Google Apps Script integration
      /*
      google.script.run
        .withSuccessHandler((result) => {
          utils.hideLoader();
          callback(result);
          utils.showMessage('Operation completed successfully');
        })
        .withFailureHandler((error) => {
          utils.hideLoader();
          utils.showMessage('Error: ' + error.message, true);
        })[functionName](args);
      */
    }

    // Navigation Function
    function navigateTo(pageId) {
      // Hide all sections
      document.querySelectorAll('main > section').forEach(section => {
        section.classList.add('hidden');
      });
      
      // Show selected section
      document.getElementById(pageId).classList.remove('hidden');
      
      // Update header title
      const headerTitle = document.getElementById('header-title');
      if (pageId === 'keywords-page') {
        headerTitle.textContent = 'Keywords';
      } else if (pageId === 'templates-page') {
        headerTitle.textContent = 'Templates';
      }
      
      // Update navigation button styles
      document.querySelectorAll('.nav-btn').forEach(btn => {
        btn.classList.remove('border-blue-500', 'bg-slate-700');
        btn.classList.add('border-transparent');
      });
      
      if (pageId === 'keywords-page') {
        document.getElementById('nav-keywords').classList.add('border-blue-500', 'bg-slate-700');
        document.getElementById('nav-keywords').classList.remove('border-transparent');
      } else if (pageId === 'templates-page') {
        document.getElementById('nav-templates').classList.add('border-blue-500', 'bg-slate-700');
        document.getElementById('nav-templates').classList.remove('border-transparent');
      }
    }

    // Render Keywords
    function renderKeywords() {
      const keywordList = document.getElementById('keyword-list');
      keywordList.innerHTML = '';
      
      state.keywords.forEach((keyword, index) => {
        const colorClass = tileColors[index % tileColors.length];
        
        const tile = document.createElement('div');
        tile.className = `${colorClass} rounded-lg p-4 relative shadow-lg hover:shadow-xl transition-all duration-300 transform hover:scale-105 cursor-pointer`;
        
        tile.innerHTML = `
          <p class="font-semibold text-white pr-6 break-words">${keyword}</p>
          <button class="delete-keyword-btn absolute top-2 right-2 bg-black bg-opacity-30 hover:bg-opacity-50 rounded-full w-6 h-6 flex items-center justify-center transition-all" data-keyword="${keyword}">
            <span class="text-white text-sm font-bold">×</span>
          </button>
        `;
        
        keywordList.appendChild(tile);
      });
    }

    // Render Templates
    function renderTemplates() {
      state.templates.forEach((template, index) => {
        const textarea = document.getElementById(`template-${index + 1}`);
        if (textarea) {
          textarea.value = template;
        }
      });
    }

    // Initialize App
    document.addEventListener('DOMContentLoaded', () => {
      // Load initial data
      serverCall('getInitialData', null, (data) => {
        state.keywords = data.keywords;
        state.templates = data.templates;
        renderKeywords();
        renderTemplates();
      });

      // Navigation Event Listeners
      document.getElementById('nav-keywords').addEventListener('click', () => {
        navigateTo('keywords-page');
      });

      document.getElementById('nav-templates').addEventListener('click', () => {
        navigateTo('templates-page');
      });

      // Add Keyword Form
      document.getElementById('add-keyword-form').addEventListener('submit', (e) => {
        e.preventDefault();
        const input = document.getElementById('new-keyword-input');
        const newKeyword = input.value.trim();
        
        if (newKeyword) {
          serverCall('addKeyword', newKeyword, () => {
            state.keywords.push(newKeyword);
            renderKeywords();
            input.value = '';
          });
        }
      });

      // Delete Keyword (Event Delegation)
      document.getElementById('keyword-list').addEventListener('click', (e) => {
        if (e.target.closest('.delete-keyword-btn')) {
          const btn = e.target.closest('.delete-keyword-btn');
          const keywordToDelete = btn.dataset.keyword;
          
          serverCall('deleteKeyword', keywordToDelete, () => {
            state.keywords = state.keywords.filter(k => k !== keywordToDelete);
            renderKeywords();
          });
        }
      });

      // Save Templates Form
      document.getElementById('templates-form').addEventListener('submit', (e) => {
        e.preventDefault();
        
        const templatesArray = [];
        for (let i = 1; i <= 10; i++) {
          const textarea = document.getElementById(`template-${i}`);
          templatesArray.push(textarea.value);
        }
        
        serverCall('saveTemplates', templatesArray, () => {
          state.templates = templatesArray;
        });
      });
    });
  </script>
