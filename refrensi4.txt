      // --- RENDER TRANSACTIONS ---
      function renderTransactions(transactions) {
        if (!transactions || transactions.length === 0) {
          transactionsList.innerHTML = '<p class="text-center text-slate-500 mt-8">Tidak ada transaksi yang ditemukan.</p>';
          return;
        }
        
        let html = '<div class="space-y-3">';
        
        transactions.forEach((transaction) => {
          const date = transaction.tanggal;
          const amount = parseFloat(transaction.nominal || transaction.jumlah || 0);
          const formattedAmount = utils.formatCurrency(Math.abs(amount));
          
          let typeClass = 'bg-gray-50 border-gray-200';
          let typeIcon = '💰';
          let typeText = transaction.jenis || 'Unknown';
          
          if (transaction.jenis === 'Pemasukan' || transaction.jenis === 'Masuk') {
            typeClass = 'bg-green-50 border-green-200';
            typeIcon = '📈';
            typeText = 'Pemasukan';
          } else if (transaction.jenis === 'Pengeluaran' || transaction.jenis === 'Keluar') {
            typeClass = 'bg-red-50 border-red-200';
            typeIcon = '📉';
            typeText = 'Pengeluaran';
          } else if (transaction.jenis === 'Transfer') {
            typeClass = 'bg-blue-50 border-blue-200';
            typeIcon = '🔄';
          }
          
          html += `
            <div class="${typeClass} border rounded-lg p-3 overflow-hidden">
              <!-- Header dengan icon, type, dan nominal -->
              <div class="flex justify-between items-center mb-3">
                <div class="flex items-center gap-2 min-w-0 flex-1">
                  <span class="text-lg flex-shrink-0">${typeIcon}</span>
                  <span class="font-semibold text-gray-800 text-sm truncate">${typeText}</span>
                  <span class="text-xs text-gray-500 flex-shrink-0">${date}</span>
                </div>
                <div class="flex-shrink-0 ml-2">
                  <div class="font-bold text-sm ${transaction.jenis === 'Pemasukan' || transaction.jenis === 'Masuk' ? 'text-green-600' : transaction.jenis === 'Pengeluaran' || transaction.jenis === 'Keluar' ? 'text-red-600' : 'text-blue-600'}">
                    ${transaction.jenis === 'Pemasukan' || transaction.jenis === 'Masuk' ? '+' : transaction.jenis === 'Pengeluaran' || transaction.jenis === 'Keluar' ? '-' : ''}${formattedAmount}
                  </div>
                </div>
              </div>
              
              <!-- Detail transaksi -->
              <div class="space-y-1 mb-3">
                <div class="text-xs text-gray-600">
                  <span class="font-medium">Kategori:</span> <span class="text-gray-700">${transaction.kategori || 'Tidak ada'}</span>
                </div>
                <div class="text-xs text-gray-600">
                  <span class="font-medium">Deskripsi:</span> <span class="text-gray-700">${transaction.deskripsi || 'Tidak ada'}</span>
                </div>
                ${transaction.jenis === 'Transfer' ? 
                  `<div class="text-xs text-gray-600">
                    <span class="font-medium">Transfer:</span> <span class="text-gray-700">${transaction.dompetDari || transaction.dompet_dari || 'Tidak ada'} → ${transaction.dompetKe || transaction.dompet_ke || 'Tidak ada'}</span>
                  </div>` : 
                  `<div class="text-xs text-gray-600">
                    <span class="font-medium">Dompet:</span> <span class="text-gray-700">${transaction.dompetDari || transaction.dompetKe || transaction.dompet_dari || transaction.dompet_ke || 'Tidak ada'}</span>
                  </div>`
                }
                ${transaction.catatan ? 
                  `<div class="text-xs text-gray-500 italic">
                    <span class="font-medium">Catatan:</span> ${transaction.catatan}
                  </div>` : ''
                }
              </div>
              
              <!-- Action buttons -->
              <div class="flex gap-2 pt-2 border-t border-gray-200">
                <button onclick="editTransaction(${transaction.id})" class="flex-1 bg-blue-500 hover:bg-blue-600 text-white text-xs font-medium py-2 px-3 rounded-lg transition duration-200">

                  Hapus
                </button>
              </div>
            </div>
          `;

        // Clear custom date inputs
        startDateInput.value = '';
        endDateInput.value = '';
        
        // Load transactions
        loadTransactionsByFilter(filterType);
        updateFilterTitle(filterType);
      };
      
      const handleDateRangeFilter = () => {
        const startDate = startDateInput.value;
        const endDate = endDateInput.value;
        
        if (!startDate || !endDate) {
          utils.showMessage('Silakan pilih tanggal mulai dan akhir', 'error');
          return;
        }
        
        if (new Date(startDate) > new Date(endDate)) {
          utils.showMessage('Tanggal mulai tidak boleh lebih besar dari tanggal akhir', 'error');
          return;
        }
        
        customDateRange = { start: startDate, end: endDate };
        currentFilter = null;
        
        // Remove active state from all filter buttons
        filterBtns.forEach(btn => btn.classList.remove('active'));
        
        // Load transactions
        loadTransactionsByDateRange(startDate, endDate);
        updateFilterTitle(null, {
          start: new Date(startDate).toLocaleDateString('id-ID'),
          end: new Date(endDate).toLocaleDateString('id-ID')
        });
      };
      
      // --- DEBUG FUNCTIONS ---
      const runDebugTests = async () => {
        utils.showMessage('Menjalankan debug tests...', 'info');
        utils.showLoader();
        
        try {
          const results = await Promise.all([
            serverCall.call('cleanTransactionSheet'),
            serverCall.call('validateSheetIntegrity'),
            serverCall.call('runComprehensiveTest')
          ]);
          
          const [cleanResult, integrityResult, comprehensiveResult] = results;
          
          let message = 'Debug Tests Completed:\n';
          message += `- Clean Sheet: ${cleanResult.status}\n`;
          message += `- Integrity Check: ${integrityResult.status}\n`;
          message += `- Comprehensive Test: ${comprehensiveResult.status}\n`;
          
          if (integrityResult.issues?.length > 0) {
            message += `\nIssues found: ${integrityResult.issues.length}`;
          }
          
          alert(message);
          utils.showMessage('Debug tests selesai!', 'success');
          
        } catch (error) {
          utils.showMessage('Error dalam debug test: ' + error.message, 'error');
        } finally {
          utils.hideLoader();
        }
      };

      // --- EVENT LISTENERS ---
      // Filter button listeners
      filterBtns.forEach(btn => {
        btn.addEventListener('click', () => {
          const filterType = btn.dataset.filter;
          handleFilterChange(filterType);
        });
      });
      
      // Date range filter listener
      applyDateRangeBtn.addEventListener('click', handleDateRangeFilter);
      
      // Reload transactions listener
      reloadTransactionsBtn.addEventListener('click', eventHandlers.reloadCurrentTransactions);
      
      // Debug button listener
      debugBtn.addEventListener('click', runDebugTests);
      
      // Other event listeners
      confirmDeleteBtn.addEventListener('click', eventHandlers.confirmDelete);
      cancelDeleteBtn.addEventListener('click', utils.hideDeleteModal);
      cancelEditBtn.addEventListener('click', () => navigateTo('transactions-page'));
      
      // Allow Enter key for date range filter
      const handleDateRangeEnter = (e) => {
        if (e.key === 'Enter') {
          handleDateRangeFilter();
        }
      };
      
      startDateInput.addEventListener('keypress', handleDateRangeEnter);
      endDateInput.addEventListener('keypress', handleDateRangeEnter);

      // Close modal when clicking outside
      deleteModal.addEventListener('click', (e) => {
        if (e.target === deleteModal) {
          utils.hideDeleteModal();
        }
      });

      // --- APP INITIALIZATION ---
      const initializeApp = async () => {
        await loadInitialData();
        
        // Initialize transaction summary untuk hari ini
        await updateTransactionSummary('today');
      };
      
      initializeApp();
    };
  </script>
